Conversion de la nota (normativa de evaluacion del MINED):
  - Nota del corte = suma de los puntos de las actividades (0 a 100).
  - Aporte de cada corte a la nota final = nota del corte x 25 / 100.
  - Nota final = suma de los aportes de los 4 cortes (equivale al promedio de los 4).
  - Niveles de logro: AA 90-100 | AS 76-89 | AF 60-75 | AI menos de 60.

-- =====================================================================
-- SARA - Cuaderno Digital de Gestion Docente y Automatizacion del
-- Rendimiento Academico
-- Script 01: esquema de la base de datos (MySQL 8 / MariaDB 10.x)
-- La tabla nota queda como el resultado consolidado por corte: la nota del
-- corte es la SUMA de puntos (0 a 100). La nota final del año es el promedio
-- de los 4 cortes (25 % cada uno) y se calcula en la aplicacion y en las
-- =====================================================================

CREATE DATABASE IF NOT EXISTS sara
    CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE sara;
SET NAMES utf8mb4;

-- ---------------------------------------------------------------------
-- USUARIO: docentes y administradores del sistema
-- La clave se guarda como hash SHA-256 (64 caracteres hexadecimales)
-- ---------------------------------------------------------------------
CREATE TABLE usuario (
    id_usuario  INT AUTO_INCREMENT PRIMARY KEY,
    nombres     VARCHAR(80)  NOT NULL,
    apellidos   VARCHAR(80)  NOT NULL,
    usuario     VARCHAR(30)  NOT NULL UNIQUE,
    clave       CHAR(64)     NOT NULL,
    rol         ENUM('ADMINISTRADOR','DOCENTE') NOT NULL,
    activo      TINYINT(1)   NOT NULL DEFAULT 1
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- GRADO y SECCION (una seccion pertenece a un grado)
-- ---------------------------------------------------------------------
CREATE TABLE grado (
    id_grado  INT AUTO_INCREMENT PRIMARY KEY,
    nombre    VARCHAR(40) NOT NULL UNIQUE
) ENGINE=InnoDB;

CREATE TABLE seccion (
    id_seccion  INT AUTO_INCREMENT PRIMARY KEY,
    id_grado    INT NOT NULL,
    nombre      VARCHAR(10) NOT NULL,
    UNIQUE (id_grado, nombre),
    FOREIGN KEY (id_grado) REFERENCES grado(id_grado)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- ESTUDIANTE: lo registra un usuario (relacion "registra")
-- ---------------------------------------------------------------------
CREATE TABLE estudiante (
    id_estudiante    INT AUTO_INCREMENT PRIMARY KEY,
    codigo           VARCHAR(20)  NOT NULL UNIQUE,
    nombres          VARCHAR(80)  NOT NULL,
    apellidos        VARCHAR(80)  NOT NULL,
    sexo             ENUM('M','F') NOT NULL,
    fecha_nacimiento DATE,
    direccion        VARCHAR(200),
    telefono         VARCHAR(20),
    id_usuario       INT NOT NULL,
    FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- ASIGNATURA y CORTE
-- ---------------------------------------------------------------------
CREATE TABLE asignatura (
    id_asignatura  INT AUTO_INCREMENT PRIMARY KEY,
    nombre         VARCHAR(60) NOT NULL,
    regimen        VARCHAR(30) NOT NULL,
    UNIQUE (nombre, regimen)
) ENGINE=InnoDB;

CREATE TABLE corte (
    id_corte     INT AUTO_INCREMENT PRIMARY KEY,
    nombre       VARCHAR(30) NOT NULL UNIQUE,
    descripcion  VARCHAR(100)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- MATRICULA: une al estudiante con un grado y seccion en un anio lectivo
-- Dar de baja a un estudiante = cambiar el estado (no se borra)
-- ---------------------------------------------------------------------
CREATE TABLE matricula (
    id_matricula     INT AUTO_INCREMENT PRIMARY KEY,
    id_estudiante    INT NOT NULL,
    id_grado         INT NOT NULL,
    id_seccion       INT NOT NULL,
    anio_lectivo     SMALLINT NOT NULL,
    modalidad        VARCHAR(30) NOT NULL DEFAULT 'Regular',
    fecha_matricula  DATE NOT NULL,
    estado           ENUM('ACTIVA','RETIRADA') NOT NULL DEFAULT 'ACTIVA',
    UNIQUE (id_estudiante, anio_lectivo),
    FOREIGN KEY (id_estudiante) REFERENCES estudiante(id_estudiante),
    FOREIGN KEY (id_grado)      REFERENCES grado(id_grado),
    FOREIGN KEY (id_seccion)    REFERENCES seccion(id_seccion)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- ACTIVIDAD (tabla nueva): actividad evaluativa de un docente para una
-- asignatura, grado, seccion y corte, con su porcentaje y fecha.
-- Los porcentajes de un mismo corte deben sumar 100 (lo valida la app).
-- ---------------------------------------------------------------------
CREATE TABLE actividad (
    id_actividad   INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario     INT NOT NULL,
    id_asignatura  INT NOT NULL,
    id_grado       INT NOT NULL,
    id_seccion     INT NOT NULL,
    id_corte       INT NOT NULL,
    anio_lectivo   SMALLINT NOT NULL,
    nombre         VARCHAR(80) NOT NULL,
    porcentaje     DECIMAL(5,2) NOT NULL,
    fecha          DATE,
    CHECK (porcentaje > 0 AND porcentaje <= 100),
    FOREIGN KEY (id_usuario)    REFERENCES usuario(id_usuario),
    FOREIGN KEY (id_asignatura) REFERENCES asignatura(id_asignatura),
    FOREIGN KEY (id_grado)      REFERENCES grado(id_grado),
    FOREIGN KEY (id_seccion)    REFERENCES seccion(id_seccion),
    FOREIGN KEY (id_corte)      REFERENCES corte(id_corte)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- CALIFICACION (tabla nueva): puntaje de un estudiante en una actividad
-- ---------------------------------------------------------------------
CREATE TABLE calificacion (
    id_calificacion  INT AUTO_INCREMENT PRIMARY KEY,
    id_matricula     INT NOT NULL,
    id_actividad     INT NOT NULL,
    puntaje          DECIMAL(5,2) NOT NULL,
    UNIQUE (id_matricula, id_actividad),
    CHECK (puntaje >= 0 AND puntaje <= 100),
    FOREIGN KEY (id_matricula) REFERENCES matricula(id_matricula),
    FOREIGN KEY (id_actividad) REFERENCES actividad(id_actividad)
        ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- NOTA: resultado consolidado por matricula, asignatura y corte
-- calif_cuantitativa = suma de puntos de las actividades del corte (0 a 100)
-- calif_cualitativa: AA (90-100), AS (76-89), AF (60-75), AI (< 60)
-- ---------------------------------------------------------------------
CREATE TABLE nota (
    id_nota             INT AUTO_INCREMENT PRIMARY KEY,
    id_matricula        INT NOT NULL,
    id_asignatura       INT NOT NULL,
    id_corte            INT NOT NULL,
    calif_cuantitativa  DECIMAL(5,2),
    calif_cualitativa   ENUM('AA','AS','AF','AI'),
    UNIQUE (id_matricula, id_asignatura, id_corte),
    FOREIGN KEY (id_matricula)  REFERENCES matricula(id_matricula),
    FOREIGN KEY (id_asignatura) REFERENCES asignatura(id_asignatura),
    FOREIGN KEY (id_corte)      REFERENCES corte(id_corte)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- ASISTENCIA
-- ---------------------------------------------------------------------
CREATE TABLE asistencia (
    id_asistencia  INT AUTO_INCREMENT PRIMARY KEY,
    id_matricula   INT NOT NULL,
    id_asignatura  INT NOT NULL,
    id_corte       INT NOT NULL,
    fecha          DATE NOT NULL,
    estado         ENUM('PRESENTE','AUSENTE','TARDE','JUSTIFICADA') NOT NULL,
    FOREIGN KEY (id_matricula)  REFERENCES matricula(id_matricula),
    FOREIGN KEY (id_asignatura) REFERENCES asignatura(id_asignatura),
    FOREIGN KEY (id_corte)      REFERENCES corte(id_corte)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- RENDIMIENTO: resumen de rendimiento academico por grado
-- ---------------------------------------------------------------------
CREATE TABLE rendimiento (
    id_rendimiento     INT AUTO_INCREMENT PRIMARY KEY,
    id_grado           INT NOT NULL,
    anio_lectivo       SMALLINT NOT NULL,
    modalidad          VARCHAR(30) NOT NULL DEFAULT 'Regular',
    matricula_inicial  INT NOT NULL DEFAULT 0,
    matricula_actual   INT NOT NULL DEFAULT 0,
    aprobados_finales  INT NOT NULL DEFAULT 0,
    aprobados_1_2      INT NOT NULL DEFAULT 0,
    aprobados_3_mas    INT NOT NULL DEFAULT 0,
    pct_retencion      DECIMAL(5,2) NOT NULL DEFAULT 0,
    UNIQUE (id_grado, anio_lectivo, modalidad),
    FOREIGN KEY (id_grado) REFERENCES grado(id_grado)
) ENGINE=InnoDB;

-- =====================================================================
-- SARA - Script 02: datos de prueba (OPCIONAL)
-- =====================================================================
USE sara;
SET NAMES utf8mb4;

-- Estudiantes registrados por el usuario "docente"
INSERT INTO estudiante (codigo, nombres, apellidos, sexo, fecha_nacimiento, id_usuario)
SELECT v.codigo, v.nombres, v.apellidos, v.sexo, v.fecha, u.id_usuario
FROM usuario u
JOIN (
    SELECT 'E-001' AS codigo, 'Ana'    AS nombres, 'Lopez'    AS apellidos, 'F' AS sexo, '2013-03-14' AS fecha UNION ALL
    SELECT 'E-002', 'Carlos', 'Mendoza',  'M', '2013-07-02' UNION ALL
    SELECT 'E-003', 'Sofía',  'Rivas',    'F', '2012-11-20' UNION ALL
    SELECT 'E-004', 'Luis',   'Castillo', 'M', '2013-01-09' UNION ALL
    SELECT 'E-005', 'María',  'Gómez',    'F', '2013-05-27'
) v
WHERE u.usuario = 'docente';

-- Matricula 2026 en Septimo grado, seccion A
INSERT INTO matricula (id_estudiante, id_grado, id_seccion, anio_lectivo, fecha_matricula)
SELECT e.id_estudiante, g.id_grado, s.id_seccion, 2026, '2026-02-01'
FROM estudiante e
JOIN grado g   ON g.nombre = 'Séptimo grado'
JOIN seccion s ON s.id_grado = g.id_grado AND s.nombre = 'A'
WHERE e.codigo IN ('E-001', 'E-002', 'E-003', 'E-004', 'E-005');

-- Actividades de Matematica (secundaria): las mismas 3 en los 4 cortes
INSERT INTO actividad (id_usuario, id_asignatura, id_grado, id_seccion, id_corte,
                       anio_lectivo, nombre, porcentaje, fecha)
SELECT u.id_usuario, a.id_asignatura, g.id_grado, s.id_seccion, c.id_corte,
       2026, v.nombre, v.porcentaje, v.fecha
FROM usuario u
JOIN asignatura a ON a.nombre = 'Matemática' AND a.regimen = 'Secundaria'
JOIN grado g      ON g.nombre = 'Séptimo grado'
JOIN seccion s    ON s.id_grado = g.id_grado AND s.nombre = 'A'
JOIN corte c
JOIN (
    SELECT 'Quiz 1' AS nombre, 20.00 AS porcentaje, '2026-03-10' AS fecha UNION ALL
    SELECT 'Tarea',  30.00, '2026-03-20' UNION ALL
    SELECT 'Examen', 50.00, '2026-04-05'
) v
WHERE u.usuario = 'docente';

-- Calificaciones: cada estudiante tiene un rendimiento base (fraccion de los
-- puntos maximos) que varia un poco por actividad y por corte.
-- puntaje = maximo de la actividad x fraccion  (nunca pasa del maximo)
INSERT INTO calificacion (id_matricula, id_actividad, puntaje)
SELECT m.id_matricula, a.id_actividad,
       ROUND(a.porcentaje * LEAST(1, GREATEST(0, p.base + v.ajuste_actividad + w.ajuste_corte)), 2)
FROM (
    SELECT 'E-001' AS codigo, 0.95 AS base UNION ALL
    SELECT 'E-002', 0.82 UNION ALL
    SELECT 'E-003', 0.70 UNION ALL
    SELECT 'E-004', 0.52 UNION ALL
    SELECT 'E-005', 0.88
) p
JOIN estudiante e ON e.codigo = p.codigo
JOIN matricula m  ON m.id_estudiante = e.id_estudiante AND m.anio_lectivo = 2026
JOIN actividad a  ON a.id_grado = m.id_grado AND a.id_seccion = m.id_seccion
                 AND a.anio_lectivo = m.anio_lectivo
JOIN asignatura asig ON asig.id_asignatura = a.id_asignatura
                    AND asig.nombre = 'Matemática' AND asig.regimen = 'Secundaria'
JOIN corte c      ON c.id_corte = a.id_corte
JOIN (
    SELECT 'Quiz 1' AS nombre, -0.03 AS ajuste_actividad UNION ALL
    SELECT 'Tarea',   0.02 UNION ALL
    SELECT 'Examen', -0.01
) v ON v.nombre = a.nombre
JOIN (
    SELECT 'I Corte' AS corte, 0.00 AS ajuste_corte UNION ALL
    SELECT 'II Corte',  0.02 UNION ALL
    SELECT 'III Corte', -0.03 UNION ALL
    SELECT 'IV Corte',  0.01
) w ON w.corte = c.nombre
WHERE NOT (p.codigo = 'E-005' AND c.nombre = 'IV Corte');

-- Nota de cada corte = suma de los puntos de sus actividades
INSERT INTO nota (id_matricula, id_asignatura, id_corte, calif_cuantitativa, calif_cualitativa)
SELECT t.id_matricula, t.id_asignatura, t.id_corte, t.total,
       CASE WHEN t.total >= 90 THEN 'AA'
            WHEN t.total >= 76 THEN 'AS'
            WHEN t.total >= 60 THEN 'AF'
            ELSE 'AI' END
FROM (SELECT c.id_matricula, a.id_asignatura, a.id_corte, ROUND(SUM(c.puntaje), 2) AS total
      FROM calificacion c
      JOIN actividad a ON a.id_actividad = c.id_actividad
      GROUP BY c.id_matricula, a.id_asignatura, a.id_corte) t;

-- ---------------------------------------------------------------------
-- Consulta de prueba: nota de cada corte y nota final por estudiante
-- nota final = promedio de los 4 cortes (cada corte aporta 25 %)
-- ---------------------------------------------------------------------
SELECT e.codigo,
       CONCAT(e.nombres, ' ', e.apellidos) AS estudiante,
       MAX(CASE WHEN c.nombre = 'I Corte'   THEN n.calif_cuantitativa END) AS i_corte,
       MAX(CASE WHEN c.nombre = 'II Corte'  THEN n.calif_cuantitativa END) AS ii_corte,
       MAX(CASE WHEN c.nombre = 'III Corte' THEN n.calif_cuantitativa END) AS iii_corte,
       MAX(CASE WHEN c.nombre = 'IV Corte'  THEN n.calif_cuantitativa END) AS iv_corte,
       CASE WHEN COUNT(*) = 4 THEN ROUND(SUM(n.calif_cuantitativa) / 4, 2) END AS nota_final
FROM nota n
JOIN corte c      ON c.id_corte = n.id_corte
JOIN matricula m  ON m.id_matricula = n.id_matricula
JOIN estudiante e ON e.id_estudiante = m.id_estudiante
WHERE e.codigo LIKE 'E-00_'
GROUP BY e.id_estudiante, e.codigo, e.nombres, e.apellidos
ORDER BY e.codigo;

-- =====================================================================
-- SARA - Script 03: consultas SQL principales
-- Son las mismas consultas que usa el sistema (en las clases DAO).
-- Antes de ejecutarlas se eligen los datos del grupo en las variables.
-- Requiere MySQL 8 o MariaDB 10.2 o superior (usa RANK()).
-- =====================================================================
USE sara;
SET NAMES utf8mb4;

-- Variables: cambiar segun el grupo que se quiere consultar
SET @anio       = 2026;
SET @id_grado   = (SELECT id_grado FROM grado WHERE nombre = 'Séptimo grado');
SET @id_seccion = (SELECT id_seccion FROM seccion WHERE id_grado = @id_grado AND nombre = 'A');
SET @id_asig    = (SELECT id_asignatura FROM asignatura WHERE nombre = 'Matemática' AND regimen = 'Secundaria');
SET @id_corte   = (SELECT id_corte FROM corte WHERE nombre = 'I Corte');

-- ---------------------------------------------------------------------
-- 1) Estudiantes activos por grupo (grado y seccion) en un anio
-- ---------------------------------------------------------------------
SELECT g.nombre AS grado, s.nombre AS seccion, COUNT(m.id_matricula) AS estudiantes_activos
FROM seccion s
JOIN grado g ON g.id_grado = s.id_grado
LEFT JOIN matricula m ON m.id_seccion = s.id_seccion
                     AND m.anio_lectivo = @anio AND m.estado = 'ACTIVA'
GROUP BY g.id_grado, g.nombre, s.nombre
HAVING COUNT(m.id_matricula) > 0
ORDER BY g.id_grado, s.nombre;

-- ---------------------------------------------------------------------
-- 2) Actividades de un corte y suma de sus puntos maximos (debe ser 100)
-- ---------------------------------------------------------------------
SELECT a.nombre AS actividad, a.porcentaje, a.fecha
FROM actividad a
WHERE a.id_grado = @id_grado AND a.id_seccion = @id_seccion AND a.id_asignatura = @id_asig
  AND a.id_corte = @id_corte AND a.anio_lectivo = @anio
ORDER BY a.fecha, a.id_actividad;

-- Cortes cuyos puntos maximos NO suman 100 (para revisar configuraciones incompletas)
SELECT g.nombre AS grado, s.nombre AS seccion, asig.nombre AS asignatura, c.nombre AS corte,
       SUM(a.porcentaje) AS suma_porcentajes
FROM actividad a
JOIN grado g       ON g.id_grado = a.id_grado
JOIN seccion s     ON s.id_seccion = a.id_seccion
JOIN asignatura asig ON asig.id_asignatura = a.id_asignatura
JOIN corte c       ON c.id_corte = a.id_corte
WHERE a.anio_lectivo = @anio
GROUP BY a.id_usuario, g.nombre, s.nombre, asig.nombre, c.nombre, a.id_grado, a.id_seccion,
         a.id_asignatura, a.id_corte
HAVING SUM(a.porcentaje) <> 100;

-- ---------------------------------------------------------------------
-- 3) Nota del corte y nivel de logro de cada estudiante
--    nota del corte = suma de los puntos de sus actividades (0 a 100)
--    aporte a la final = nota del corte x 25 / 100
-- ---------------------------------------------------------------------
SELECT e.codigo,
       CONCAT(e.nombres, ' ', e.apellidos) AS estudiante,
       ROUND(SUM(c.puntaje), 2) AS nota_del_corte,
       ROUND(SUM(c.puntaje) * 25 / 100, 2) AS aporte_a_la_final,
       CASE
           WHEN SUM(c.puntaje) >= 90 THEN 'AA'
           WHEN SUM(c.puntaje) >= 76 THEN 'AS'
           WHEN SUM(c.puntaje) >= 60 THEN 'AF'
           ELSE 'AI'
       END AS nivel_logro
FROM calificacion c
JOIN actividad a  ON a.id_actividad = c.id_actividad
JOIN matricula m  ON m.id_matricula = c.id_matricula
JOIN estudiante e ON e.id_estudiante = m.id_estudiante
WHERE a.id_grado = @id_grado AND a.id_seccion = @id_seccion AND a.id_asignatura = @id_asig
  AND a.id_corte = @id_corte AND a.anio_lectivo = @anio AND m.estado = 'ACTIVA'
GROUP BY e.id_estudiante, e.codigo, e.nombres, e.apellidos
ORDER BY e.apellidos, e.nombres;

-- ---------------------------------------------------------------------
-- 4) Ranking del grupo (los empates comparten posicion)
-- ---------------------------------------------------------------------
SELECT RANK() OVER (ORDER BY n.calif_cuantitativa DESC) AS posicion,
       e.codigo,
       CONCAT(e.nombres, ' ', e.apellidos) AS estudiante,
       n.calif_cuantitativa AS nota_del_corte,
       n.calif_cualitativa  AS nivel
FROM nota n
JOIN matricula m  ON m.id_matricula = n.id_matricula
JOIN estudiante e ON e.id_estudiante = m.id_estudiante
WHERE m.id_grado = @id_grado AND m.id_seccion = @id_seccion AND m.anio_lectivo = @anio
  AND m.estado = 'ACTIVA' AND n.id_asignatura = @id_asig AND n.id_corte = @id_corte
ORDER BY posicion, e.apellidos;

-- ---------------------------------------------------------------------
-- 5) Alumnos en riesgo academico: nota del corte menor a 60
-- ---------------------------------------------------------------------
SELECT e.codigo, CONCAT(e.nombres, ' ', e.apellidos) AS estudiante,
       asig.nombre AS asignatura, n.calif_cuantitativa AS nota_del_corte, n.calif_cualitativa AS nivel
FROM nota n
JOIN matricula m     ON m.id_matricula = n.id_matricula
JOIN estudiante e    ON e.id_estudiante = m.id_estudiante
JOIN asignatura asig ON asig.id_asignatura = n.id_asignatura
WHERE m.id_grado = @id_grado AND m.id_seccion = @id_seccion AND m.anio_lectivo = @anio
  AND m.estado = 'ACTIVA' AND n.id_corte = @id_corte AND n.calif_cuantitativa < 60
ORDER BY e.apellidos, e.nombres, asig.nombre;

-- ---------------------------------------------------------------------
-- 6) Boletin de la seccion: nota de cada corte, acumulado, nota final y letra
--    nota final = promedio de los 4 cortes (25 % cada uno); solo existe si
--    el estudiante ya tiene los 4 cortes con nota. Si no, queda NULL (En proceso).
--    Para un solo estudiante agregar:  AND e.codigo = 'E-001'
-- ---------------------------------------------------------------------
SELECT t.codigo, t.estudiante, t.asignatura,
       t.i_corte, t.ii_corte, t.iii_corte, t.iv_corte,
       t.acumulado, t.nota_final,
       CASE WHEN t.nota_final IS NULL THEN NULL
            WHEN t.nota_final >= 90 THEN 'AA'
            WHEN t.nota_final >= 76 THEN 'AS'
            WHEN t.nota_final >= 60 THEN 'AF'
            ELSE 'AI' END AS nivel,
       CASE WHEN t.nota_final IS NULL THEN NULL
            WHEN t.nota_final >= 60 THEN 'Aprobada'
            ELSE 'Reprobada' END AS resultado
FROM (SELECT e.codigo,
             CONCAT(e.nombres, ' ', e.apellidos) AS estudiante,
             e.apellidos, e.nombres,
             asig.nombre AS asignatura,
             MAX(CASE WHEN c.nombre = 'I Corte'   THEN n.calif_cuantitativa END) AS i_corte,
             MAX(CASE WHEN c.nombre = 'II Corte'  THEN n.calif_cuantitativa END) AS ii_corte,
             MAX(CASE WHEN c.nombre = 'III Corte' THEN n.calif_cuantitativa END) AS iii_corte,
             MAX(CASE WHEN c.nombre = 'IV Corte'  THEN n.calif_cuantitativa END) AS iv_corte,
             ROUND(SUM(n.calif_cuantitativa) / 4, 2) AS acumulado,
             CASE WHEN COUNT(*) = 4 THEN ROUND(SUM(n.calif_cuantitativa) / 4, 2) END AS nota_final
      FROM nota n
      JOIN corte c         ON c.id_corte = n.id_corte
      JOIN matricula m     ON m.id_matricula = n.id_matricula
      JOIN estudiante e    ON e.id_estudiante = m.id_estudiante
      JOIN asignatura asig ON asig.id_asignatura = n.id_asignatura
      WHERE m.id_grado = @id_grado AND m.id_seccion = @id_seccion AND m.anio_lectivo = @anio
        AND m.estado = 'ACTIVA'
      GROUP BY e.id_estudiante, e.codigo, e.nombres, e.apellidos, asig.id_asignatura, asig.nombre) t
ORDER BY t.apellidos, t.nombres, t.asignatura;

-- ---------------------------------------------------------------------
-- 7) Promedio del grupo (de las notas de corte) por asignatura y corte
-- ---------------------------------------------------------------------
SELECT asig.nombre AS asignatura, c.nombre AS corte,
       ROUND(AVG(n.calif_cuantitativa), 2) AS promedio_del_grupo,
       COUNT(*) AS estudiantes_con_nota
FROM nota n
JOIN matricula m     ON m.id_matricula = n.id_matricula
JOIN asignatura asig ON asig.id_asignatura = n.id_asignatura
JOIN corte c         ON c.id_corte = n.id_corte
WHERE m.id_grado = @id_grado AND m.id_seccion = @id_seccion AND m.anio_lectivo = @anio
  AND m.estado = 'ACTIVA'
GROUP BY asig.id_asignatura, asig.nombre, c.id_corte, c.nombre
ORDER BY asig.nombre, c.id_corte;

-- ---------------------------------------------------------------------
-- 8) Cantidad de estudiantes por nivel de logro (distribucion del grupo)
-- ---------------------------------------------------------------------
SELECT n.calif_cualitativa AS nivel, COUNT(*) AS estudiantes
FROM nota n
JOIN matricula m ON m.id_matricula = n.id_matricula
WHERE m.id_grado = @id_grado AND m.id_seccion = @id_seccion AND m.anio_lectivo = @anio
  AND m.estado = 'ACTIVA' AND n.id_asignatura = @id_asig AND n.id_corte = @id_corte
GROUP BY n.calif_cualitativa
ORDER BY FIELD(n.calif_cualitativa, 'AA', 'AS', 'AF', 'AI');

-- ---------------------------------------------------------------------
-- 9) Rendimiento del grado: asignaturas reprobadas por estudiante
--    Una asignatura se reprueba si su nota final (promedio de los 4 cortes)
--    es menor a 60. Solo cuentan las asignaturas con los 4 cortes ya calificados.
-- ---------------------------------------------------------------------
SELECT m.modalidad,
       COUNT(*) AS estudiantes_con_notas,
       SUM(CASE WHEN r.reprobadas = 0 THEN 1 ELSE 0 END)               AS aprobaron_todas,
       SUM(CASE WHEN r.reprobadas BETWEEN 1 AND 2 THEN 1 ELSE 0 END)   AS reprobaron_1_o_2,
       SUM(CASE WHEN r.reprobadas >= 3 THEN 1 ELSE 0 END)              AS reprobaron_3_o_mas
FROM matricula m
JOIN (SELECT x.id_matricula,
             SUM(CASE WHEN x.nota_final < 60 THEN 1 ELSE 0 END) AS reprobadas
      FROM (SELECT id_matricula, id_asignatura,
                   ROUND(SUM(calif_cuantitativa) / 4, 2) AS nota_final
            FROM nota GROUP BY id_matricula, id_asignatura
            HAVING COUNT(*) = 4) x
      GROUP BY x.id_matricula) r ON r.id_matricula = m.id_matricula
WHERE m.id_grado = @id_grado AND m.anio_lectivo = @anio AND m.estado = 'ACTIVA'
GROUP BY m.modalidad;

-- ---------------------------------------------------------------------
-- 10) Retencion: matricula inicial vs matricula actual por grado
-- ---------------------------------------------------------------------
SELECT g.nombre AS grado,
       COUNT(*) AS matricula_inicial,
       SUM(CASE WHEN m.estado = 'ACTIVA' THEN 1 ELSE 0 END) AS matricula_actual,
       ROUND(SUM(CASE WHEN m.estado = 'ACTIVA' THEN 1 ELSE 0 END) * 100 / COUNT(*), 2) AS pct_retencion
FROM matricula m
JOIN grado g ON g.id_grado = m.id_grado
WHERE m.anio_lectivo = @anio
GROUP BY g.id_grado, g.nombre
ORDER BY g.id_grado;

-- =====================================================================
-- SARA - Script 04: MIGRACION a la version 5 (OPCIONAL)
-- Para quien YA tiene la base "sara" creada con el script 01 de la v4
-- y no quiere perder sus datos. Si va a crear la base desde cero, NO
-- hace falta: el script 01 nuevo ya trae todo.
--
-- Que hace:
--   1) Agrega el IV Corte (si no existe).
--   2) Convierte los datos de ejemplo de Septimo A (Matematica) de la
--      escala vieja (0 a 100) a puntos, solo si todavia no estan convertidos.
--   3) Recalcula la tabla nota: la nota de cada corte pasa a ser la
--      SUMA de puntos (antes era un promedio ponderado).
--
-- Tus puntajes de Segundo C (8, 13, 22, 20, 15...) ya estan en puntos:
-- no se tocan, solo se recalcula su total.
-- Se puede ejecutar mas de una vez sin danar nada.
-- =====================================================================
USE sara;
SET NAMES utf8mb4;
SET SQL_SAFE_UPDATES = 0;

-- 1) IV Corte
INSERT INTO corte (nombre, descripcion)
SELECT 'IV Corte', 'Cuarto corte evaluativo'
WHERE NOT EXISTS (SELECT 1 FROM corte WHERE nombre = 'IV Corte');

-- 2) Datos de ejemplo viejos (E-001 a E-005, Matematica de Septimo A):
--    tenian notas de 0 a 100 en actividades de 20 / 30 / 50 %.
--    Si alguna pasa del maximo de su actividad, se convierten todas a puntos.
SET @sin_convertir = (
    SELECT COUNT(*)
    FROM calificacion c
    JOIN actividad a     ON a.id_actividad = c.id_actividad
    JOIN asignatura asig ON asig.id_asignatura = a.id_asignatura
    JOIN grado g         ON g.id_grado = a.id_grado
    JOIN seccion s       ON s.id_seccion = a.id_seccion
    JOIN matricula m     ON m.id_matricula = c.id_matricula
    JOIN estudiante e    ON e.id_estudiante = m.id_estudiante
    WHERE asig.nombre = 'Matemática' AND asig.regimen = 'Secundaria'
      AND g.nombre = 'Séptimo grado' AND s.nombre = 'A'
      AND e.codigo IN ('E-001', 'E-002', 'E-003', 'E-004', 'E-005')
      AND a.nombre IN ('Quiz 1', 'Tarea', 'Examen')
      AND c.puntaje > a.porcentaje
);

UPDATE calificacion c
JOIN actividad a     ON a.id_actividad = c.id_actividad
JOIN asignatura asig ON asig.id_asignatura = a.id_asignatura
JOIN grado g         ON g.id_grado = a.id_grado
JOIN seccion s       ON s.id_seccion = a.id_seccion
JOIN matricula m     ON m.id_matricula = c.id_matricula
JOIN estudiante e    ON e.id_estudiante = m.id_estudiante
SET c.puntaje = ROUND(c.puntaje * a.porcentaje / 100, 2)
WHERE @sin_convertir > 0
  AND asig.nombre = 'Matemática' AND asig.regimen = 'Secundaria'
  AND g.nombre = 'Séptimo grado' AND s.nombre = 'A'
  AND e.codigo IN ('E-001', 'E-002', 'E-003', 'E-004', 'E-005')
  AND a.nombre IN ('Quiz 1', 'Tarea', 'Examen');

-- Revision: calificaciones que todavia pasan del maximo de su actividad.
-- Si esta consulta devuelve filas, son notas tuyas que hay que corregir
-- en el registro de notas (no se corrigen solas).
SELECT e.codigo, CONCAT(e.nombres, ' ', e.apellidos) AS estudiante,
       a.nombre AS actividad, a.porcentaje AS maximo, c.puntaje
FROM calificacion c
JOIN actividad a  ON a.id_actividad = c.id_actividad
JOIN matricula m  ON m.id_matricula = c.id_matricula
JOIN estudiante e ON e.id_estudiante = m.id_estudiante
WHERE c.puntaje > a.porcentaje;

-- 3) Recalcular toda la tabla nota desde las calificaciones
--    (nota del corte = suma de puntos)
DELETE FROM nota;

INSERT INTO nota (id_matricula, id_asignatura, id_corte, calif_cuantitativa, calif_cualitativa)
SELECT t.id_matricula, t.id_asignatura, t.id_corte, t.total,
       CASE WHEN t.total >= 90 THEN 'AA'
            WHEN t.total >= 76 THEN 'AS'
            WHEN t.total >= 60 THEN 'AF'
            ELSE 'AI' END
FROM (SELECT c.id_matricula, a.id_asignatura, a.id_corte, ROUND(SUM(c.puntaje), 2) AS total
      FROM calificacion c
      JOIN actividad a ON a.id_actividad = c.id_actividad
      GROUP BY c.id_matricula, a.id_asignatura, a.id_corte) t;

SET SQL_SAFE_UPDATES = 1;

-- Resultado: cuantas notas quedaron y los cortes disponibles
SELECT (SELECT COUNT(*) FROM nota)  AS notas_recalculadas,
       (SELECT GROUP_CONCAT(nombre ORDER BY id_corte SEPARATOR ', ') FROM corte) AS cortes;
